34void ClientHandler::handle() {

71bool ClientHandler::authenticateClient() {
72 // Шаг 1: Получаем логин
73 std::string login;
74 if (!receiveString(login)) {
75 logger.error("Failed to receive login");
76 return false;
77 }
78
79 // Удаляем все пробелы из имени пользователя
80 login.erase(0, login.find_first_not_of(" \t\n\r"));
81 login.erase(login.find_last_not_of(" \t\n\r") + 1);
82
83 logger.info("Received login: '" + login + "'");
84
85 // Шаг 2: Проверяем, существует ли пользователь
86 if (!clientBase.userExists(login)) {
87 logger.warning("User not found: " + login);
88 sendMessage("ERR");
89 return false;
90 }
91
92 // ИСПОЛЬЗУЕМ ПАРОЛЬ ИЗ БАЗЫ ДАННЫХ
93 std::string password = clientBase.getPassword(login);
94
95 // Шаг 3: Генерируем и отправляем соль с помощью Crypto++
96 std::string salt = Authenticator::generateSalt();
97 logger.info("Generated salt for user " + login + ": " + salt);
98
99 if (!sendSalt(salt)) {
100 logger.error("Failed to send salt");
101 return false;
102 }
103
104 // Шаг 4: Получаем хэш от клиента
105 std::string clientHash;
106
107 // Используем receiveString для хеша тоже
108 if (!receiveString(clientHash)) {
109 logger.error("Failed to receive hash");
110 return false;
111 }
112
113 logger.info("Received hash from client");
114
115 // Шаг 5: Аутентификация с помощью Crypto++
116 bool authResult = Authenticator::authenticate(password, salt, clientHash);
117
118 if (authResult) {
119 logger.info("Authentication successful for user: " + login);
120 return sendMessage("OK");
121 } else {
122 logger.warning("Authentication failed for user: " + login);
123 sendMessage("ERR");
124 return false;
125 }
126}

139bool ClientHandler::receiveVectorsAndProcess() {
140 // Шаг 6: Получаем количество векторов
141 uint32_t numVectors;
142 if (!receiveData(&numVectors, sizeof(numVectors))) {
143 logger.error("Failed to receive number of vectors");
144 return false;
145 }
146
147 numVectors = le32toh(numVectors);
148
149 if (numVectors > 1000) {
150 logger.error("Invalid number of vectors: " + std::to_string(numVectors));
151 return false;
152 }
153
154 // НЕ отправляем количество результатов - это не результат вычислений
155 // Просто переходим к обработке векторов
156
157 // Обрабатываем каждый вектор
158 for (uint32_t i = 0; i < numVectors; i++) {
159 // Шаг 7: Получаем размер вектора
160 uint32_t vectorSize;
161 if (!receiveData(&vectorSize, sizeof(vectorSize))) {
162 logger.error("Failed to receive vector size");
163 return false;
164 }
165 vectorSize = le32toh(vectorSize);
166
167 if (vectorSize > 100000) {
168 logger.error("Invalid vector size: " + std::to_string(vectorSize));
169 return false;
170 }
171
172 // Шаг 8: Получаем данные вектора
173 std::vector<uint64_t> vector(vectorSize);
174 for (uint32_t j = 0; j < vectorSize; j++) {
175 uint64_t value;
176 if (!receiveData(&value, sizeof(value))) {
177 logger.error("Failed to receive vector data");
178 return false;
179 }
180 value = le64toh(value);
181 vector[j] = value;
182 }
183
184 // Шаг 9: Вычисляем и отправляем результат
185 uint64_t result = VectorCalculator::computeProduct(vector);
186
187 result = htole64(result);
188 if (!sendData(&result, sizeof(result))) {
189 logger.error("Failed to send result");
190 return false;
191 }
192 }
193
194 return true;
195}

278bool ClientHandler::receiveString(std::string& result, size_t maxSize) {
279 char buffer[256] = {0};
280 size_t totalReceived = 0;
281
282
283 // Сначала посмотрим, что находится в буфере
284 char peekBuffer[256] = {0};
285 ssize_t peeked = recv(clientSocket, peekBuffer, sizeof(peekBuffer) - 1, MSG_PEEK);
286
287 if (peeked > 0) {
288
289 // Если мы видим полную строку без терминатора, считываем ровно столько байт
290 bool hasTerminator = false;
291 for (int i = 0; i < peeked; i++) {
292 if (peekBuffer[i] == '\0' || peekBuffer[i] == '\n' || peekBuffer[i] == '\r') {
293 hasTerminator = true;
294 break;
295 }
296 }
297
298 if (!hasTerminator && peeked > 0) {
299 // Терминатор не найден, прочитаем ровно столько байт, сколько мы просмотрели
300 ssize_t received = recv(clientSocket, buffer, peeked, 0);
301 if (received == peeked) {
302 buffer[received] = '\0'; // Добавляем нулевой терминатор
303 result = std::string(buffer);
304 return true;
305 }
306 }
307 }
308
309 // Резервный вариант: читаем побайтно
310 while (totalReceived < maxSize - 1) {
311 ssize_t received = recv(clientSocket, &buffer[totalReceived], 1, 0);
312
313 if (received <= 0) {
314 return false;
315 }
316
317
318 // Проверяем наличие различных возможных терминаторов
319 if (buffer[totalReceived] == '\0' ||
320 buffer[totalReceived] == '\n' ||
321 buffer[totalReceived] == '\r') {
322
323 buffer[totalReceived] = '\0'; // Гарантируем нулевой терминатор
324 break;
325 }
326
327 totalReceived++;
328
329 // Если мы получили 4 байта и это «пользователь», считаем, что он завершен
330 if (totalReceived == 4 && strncmp(buffer, "user", 4) == 0) {
331 buffer[4] = '\0'; // Добавляем нулевой терминатор
332 break;
333 }
334 }
335
336 result = std::string(buffer);
337 return true;
338}
