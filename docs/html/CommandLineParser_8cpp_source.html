58bool CommandLineParser::parse(int argc, char* argv[]) {
59 // Сбрасываем конфиг перед парсингом
60 config.port = 0;
61 config.clientBaseFile = "";
62 config.logFile = "";
63
64 // Сбрасываем optind для повторных вызовов getopt_long
65 optind = 1;
66
67 // Структура для длинных опций getopt_long
68 static struct option long_options[] = {
69 {"base", required_argument, 0, 'b'}, // Файл базы клиентов
70 {"log", required_argument, 0, 'l'}, // Файл журнала
71 {"port", required_argument, 0, 'p'}, // Порт сервера
72 {"help", no_argument, 0, 'h'}, // Справка
73 {0, 0, 0, 0} // Конец массива
74 };
75
76 // Строка коротких опций для getopt
77 const char* short_options = "b:l:p:h";
78
79 // Разбор опций с помощью getopt_long
80 int opt;
81 int option_index = 0;
82
83 // Используем цикл для обработки всех опций
84 while ((opt = getopt_long(argc, argv, short_options, long_options, &option_index)) != -1) {
85 switch (opt) {
86 case 'b': // Файл базы клиентов
87 config.clientBaseFile = optarg;
88 break;
89
90 case 'l': // Файл журнала
91 config.logFile = optarg;
92 break;
93
94 case 'p': // Порт сервера
95 try {
96 config.port = stoi(string(optarg));
97
98 // Проверяем допустимость порта
99 if (config.port < 1 || config.port > 65535) {
100 cerr << "Error: Port must be between 1 and 65535" << endl;
101 return false;
102 }
103 } catch (const exception&) {
104 cerr << "Error: Invalid port number: " << optarg << endl;
105 return false;
106 }
107 break;
108
109 case 'h': // Справка
110 cout << helpText;
111 return false; // Завершаем выполнение после вывода справки
112
113 case '?': // Неизвестная опция
114 // getopt_long уже выводит сообщение об ошибке
115 cerr << "\nUse -h or --help for usage information" << endl;
116 return false;
117
118 case ':': // Отсутствует аргумент для опции
119 cerr << "Error: Missing argument for option '"
120 << static_cast<char>(optopt) << "'" << endl;
121 cerr << "Use -h or --help for usage information" << endl;
122 return false;
123
124 default:
125 cerr << "Error: Unknown error parsing parameters" << endl;
126 return false;
127 }
128 }
129
130 // Проверяем наличие лишних аргументов (не опций)
131 if (optind < argc) {
132 cerr << "Error: Unknown arguments: ";
133 for (int i = optind; i < argc; i++) {
134 cerr << argv[i] << " ";
135 }
136 cerr << "\nUse -h or --help for usage information" << endl;
137 return false;
138 }
139
140 // Проверяем обязательные параметры
141 if (!validateRequiredParams()) {
142 cerr << "Use -h or --help for usage information" << endl;
143 return false;
144 }
145
146 return true;
147}
